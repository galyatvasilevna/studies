## Полный код решения

```cpp
#include <iostream>
#include <vector>
#include <chrono>
#include <algorithm>
using namespace std;
using namespace chrono;

const int INF = 1e9; // Бесконечность

class Graph {
    int V;
    vector<vector<int>> adj;

public:
    Graph(int v) : V(v), adj(v, vector<int>(v, INF)) {
        for (int i = 0; i < V; i++) adj[i][i] = 0;
    }

    // Неориентированный граф
    void addEdge(int u, int v, int w) {
        adj[u][v] = w;
        adj[v][u] = w;
    }

    // --- 1. Дейкстра O(V^2) с восстановлением пути ---
    // Возвращает массив расстояний и массив предков (prev)
    pair<vector<int>, vector<int>> dijkstra(int start) {
        vector<int> dist(V, INF), prev(V, -1);
        vector<bool> visited(V, false);
        dist[start] = 0;

        for (int i = 0; i < V; i++) {
            int u = -1;
            for (int j = 0; j < V; j++) {
                if (!visited[j] && (u == -1 || dist[j] < dist[u])) u = j;
            }
            if (dist[u] == INF) break;
            visited[u] = true;

            for (int v = 0; v < V; v++) {
                if (adj[u][v] != INF && dist[u] + adj[u][v] < dist[v]) {
                    dist[v] = dist[u] + adj[u][v];
                    prev[v] = u;
                }
            }
        }
        return { dist, prev };
    }

    // Вспомогательная функция для вывода пути
    void printPath(int start, int end, const vector<int>& prev) {
        if (end == -1) return;
        vector<int> path;
        for (int v = end; v != -1; v = prev[v]) path.push_back(v);
        reverse(path.begin(), path.end());
        cout << "path: ";
        for (int v : path) cout << v << " ";
        cout << "\n";
    }

    // ВАРИАНТ 11: Нахождение всех путей от одной вершины
    void printAllPaths(int start) {
        auto [dist, prev] = dijkstra(start);
        cout << "Dijkstra from " << start << ":\n";
        for (int i = 0; i < V; i++) {
            if (i == start) continue;
            cout << " to " << i << " (dist " << dist[i] << ") ";
            if (dist[i] == INF) cout << "unreachable\n";
            else printPath(start, i, prev);
        }
    }

    // --- 2. Флойд-Уоршелл ---
    vector<vector<int>> floydWarshall() {
        vector<vector<int>> dist = adj;
        for (int k = 0; k < V; k++) {
            for (int i = 0; i < V; i++) {
                for (int j = 0; j < V; j++) {
                    if (dist[i][k] != INF && dist[k][j] != INF && dist[i][k] + dist[k][j] < dist[i][j]) {
                        dist[i][j] = dist[i][k] + dist[k][j];
                    }
                }
            }
        }
        return dist;
    }
};

// --- 3. Сравнительный анализ ---
void runBenchmark(int N) {
    Graph g(N);
    for (int i = 0; i < N; i++) {
        for (int j = i + 1; j < N; j++) {
            if (rand() % 100 < 30) { // Вероятность ребра 30%
                g.addEdge(i, j, rand() % 50 + 1);
            }
        }
    }

    // Замер Флойда
    auto t1 = high_resolution_clock::now();
    g.floydWarshall();
    auto t2 = high_resolution_clock::now();
    auto timeFloyd = duration_cast<microseconds>(t2 - t1).count();

    // Замер Дейкстры (запускаем для всех вершин, чтобы найти все пути)
    t1 = high_resolution_clock::now();
    for (int i = 0; i < N; i++) {
        g.dijkstra(i); // Вызов O(V^2) N раз дает O(V^3), как и Флойд
    }
    t2 = high_resolution_clock::now();
    auto timeDijkstra = duration_cast<microseconds>(t2 - t1).count();

    cout << N << "\t\t" << timeDijkstra << "\t\t" << timeFloyd << "\n";
}

int main() {
    // 1. Демонстрация на маленьком графе (V=6)
    Graph g(6);
    g.addEdge(0, 1, 4); g.addEdge(0, 2, 2);
    g.addEdge(1, 2, 1); g.addEdge(1, 3, 5);
    g.addEdge(2, 3, 8); g.addEdge(2, 4, 10);
    g.addEdge(3, 4, 2); g.addEdge(3, 5, 6);
    g.addEdge(4, 5, 3);

    cout << "--- Paths from 0 (Dijkstra) ---\n";
    g.printAllPaths(0);

    cout << "\n--- Floyd-Warshall Matrix ---\n";
    auto dist = g.floydWarshall();
    for (int i = 0; i < 6; i++) {
        for (int j = 0; j < 6; j++) {
            if (dist[i][j] == INF) cout << "INF\t";
            else cout << dist[i][j] << "\t";
        }
        cout << "\n";
    }

    // 2. Сравнение (Вариант 11 требует V=50, и часть 3 требует 10, 20, 50, 100)
    cout << "\n--- Performance Benchmark (Dijkstra vs Floyd) ---\n";
    cout << "V\t\tDijkstra(All)\tFloyd-Warshall\n";
    cout << "--------------------------------------------------\n";

    vector<int> sizes = { 10, 20, 50, 100 }; // V=50 - требование варианта
    for (int N : sizes) {
        runBenchmark(N);
    }

    return 0;
}
```

## Ответы на контрольные вопросы

**1. Какие условия должны выполняться для корректной работы алгоритма Дейкстры?**  
Все веса рёбер должны быть неотрицательными.

**2. Почему алгоритм Дейкстры не работает с отрицательными весами?**  
Он использует жадную стратегию: выбирает вершину с минимальным расстоянием и фиксирует её. Отрицательное ребро может позже уменьшить это расстояние, что нарушает логику алгоритма.

**3. Какова временная сложность алгоритма Дейкстры с приоритетной очередью?**  
O((V + E) log V).

**4. В чём заключается идея алгоритма Флойда-Уоршелла?**  
Динамическое программирование: последовательно разрешаем использовать вершины 0..k в качестве промежуточных для улучшения путей между всеми парами вершин.

**5. Какова временная сложность алгоритма Флойда-Уоршелла?**  
O(V³).

**6. Как алгоритм Флойда-Уоршелла обнаруживает отрицательные циклы?**  
После завершения алгоритма проверяются диагональные элементы матрицы: если `dist[i][i] < 0`, значит, существует отрицательный цикл.

**7. В каких случаях предпочтительнее использовать алгоритм Дейкстры?**  
Когда нужны кратчайшие пути из одной вершины, граф разреженный (много вершин, мало рёбер) и нет отрицательных весов.

**8. В каких случаях предпочтительнее использовать алгоритм Флойда-Уоршелла?**  
Когда нужны кратчайшие пути между всеми парами вершин, граф плотный (много рёбер) или есть отрицательные веса (при отсутствии циклов).

**9. Как восстановить кратчайший путь после выполнения алгоритма Дейкстры?**  
С помощью массива предков `prev`, который заполняется при релаксации. Путь восстанавливается в обратном порядке от конечной вершины к начальной.

**10. Как восстановить кратчайший путь после выполнения алгоритма Флойда-Уоршелла?**  
С помощью матрицы `next` (или `prev`), где `next[i][j]` хранит следующую вершину на пути от `i` к `j`. Путь строится итеративно.

**11. Что такое релаксация в контексте алгоритмов поиска кратчайших путей?**  
Это операция обновления текущего известного расстояния до вершины `v`, если найден более короткий путь через вершину `u`: `dist[v] = min(dist[v], dist[u] + weight(u, v))`.

**12. Как можно оптимизировать алгоритм Дейкстры для разреженных графов?**  
Использовать список смежности вместо матрицы и `std::priority_queue` (или кучу Фибоначчи) для быстрого извлечения минимума.

**13. Какую память требует алгоритм Флойда-Уоршелла?**  
O(V²) для хранения матрицы расстояний.

**14. При каком количестве вершин алгоритм Флойда-Уоршелла становится неприменимым?**  
При очень больших V (например, V > 5000-10000) из-за кубической временной сложности O(V³) и квадратичной памяти.

**15. Как модифицировать алгоритм Дейкстры для нахождения второго кратчайшего пути?**  
Хранить для каждой вершины два лучших расстояния (кратчайшее и второе кратчайшее) и обновлять их в процессе релаксации.
